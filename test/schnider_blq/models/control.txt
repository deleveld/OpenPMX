$PROB Schnider
$INPUT <fromdata>
$DATA "data.csv" IGNORE=@
$SUBROUTINES ADVAN13 TOL=9
$MODEL
COMPARTMENT=(CENTRAL     DEFDOSE DEFOBS)
COMPARTMENT=(PERIPH1)
COMPARTMENT=(PERIPH2)
$PK
SIZE = WT/70.;
MU_V1 = LOG(THETA(1)) + LOG(SIZE)
V1 = EXP(MU_V1 + ETA(1))
MU_V2 = LOG(THETA(2)) + LOG(SIZE)
V2 = EXP(MU_V2 + ETA(2))
MU_V3 = LOG(THETA(3)) + LOG(SIZE)
V3 = EXP(MU_V3 + ETA(3))
MU_CL = LOG(THETA(4)) + 0.75*LOG(SIZE)
CL = EXP(MU_CL + ETA(4))
MU_Q2 = LOG(THETA(5)) + 0.75*LOG(SIZE)
Q2 = EXP(MU_Q2 + ETA(5))
MU_Q3 = LOG(THETA(6)) + 0.75*LOG(SIZE)
Q3 = EXP(MU_Q3 + ETA(6))
SDPROP=THETA(7)
SDADD=THETA(8)
S1=V1
K10 = CL/V1
K12 = Q2/V1
K21 = Q2/V2
K13 = Q3/V1
K31 = Q3/V3
$DES
DADT(1) = -(K10+K12+K13)*A(1) + K21*A(2) + K31*A(3)
DADT(2) =  K12*A(1) - K21*A(2)
DADT(3) =  K13*A(1) - K31*A(3)
$THETA
(0.1, 6, 25)
(1, 20, 125)
(25, 200, 500)
(0.1,  2, 6)
(0.1, 1, 3)
(0.1, 1, 3)
(0.1, 0.22, 1)
(1e-5 FIXED)
$OMEGA
0.1
0.1
0 FIXED
0.1
0 FIXED
0 FIXED
$SIGMA 1 FIXED
$ERROR(ONLY OBSERVATIONS)
	IPRED=A(1)/V1
	W = SQRT((IPRED * SDPROP)**2 + SDADD**2)
	Y = IPRED + W*ERR(1)
